@ExtendWith(MockitoExtension.class)
class AuthServiceTest {

    @Mock
    private UserMapper userMapper;

    @Mock
    private PasswordEncoder passwordEncoder;

    @Mock
    private JwtTokenService jwtTokenService;

    @Mock
    private LoginLogMapper loginLogMapper;

    @InjectMocks
    private AuthService authService;

    @Test
    @DisplayName("登录成功-返回Token")
    void testLoginSuccess() {
        // Given
        LoginDTO loginDTO = new LoginDTO();
        loginDTO.setUsername("admin");
        loginDTO.setPassword("admin123");

        User user = createMockUser();
        when(userMapper.selectByUsername("admin")).thenReturn(user);
        when(passwordEncoder.matches("admin123", user.getPassword())).thenReturn(true);
        when(jwtTokenService.generateToken(any())).thenReturn("mock-token");

        // When
        LoginResult result = authService.login(loginDTO);

        // Then
        assertNotNull(result);
        assertEquals("mock-token", result.getToken());
        verify(loginLogMapper).insert(any(LoginLog.class));
    }

    @Test
    @DisplayName("登录失败-用户不存在")

        // When & Then

## 四、缺陷管理

### 4.1 缺陷等级定义

| 等级 | 定义 | 示例 |
|------|------|------|
| **P0 - 致命** | 系统无法运行，核心功能完全不可用 | 应用无法启动、登录功能完全失效 |
| **P1 - 严重** | 核心功能受限，但存在替代方案 | Token验证失败、数据库连接失败 |
| **P2 - 一般** | 功能存在缺陷，但不影响主要流程 | 表单验证不完善、样式轻微偏差 |
| **P3 - 轻微** | 优化项，不影响功能 | 文案错误、提示信息不友好 |
